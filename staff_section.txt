    # ============================================
    # TAB 2: Staff Management
    # ============================================
    with tab2:
        st.markdown("### 직원 관리")
        
        # Add New Staff
        with st.expander("➕ 새 직원 추가"):
            with st.form("add_staff_form"):
                col1, col2 = st.columns(2)
                new_name = col1.text_input("이름 (필수)")
                new_car = col2.selectbox("차종 (필수)", ["SEDAN", "SUV"])
                
                col3, col4 = st.columns(2)
                new_car_number = col3.text_input("차 번호 (선택)")
                new_car_details = col4.text_input("상세 차종 (선택)")
                
                if st.form_submit_button("추가", type="primary"):
                    if not new_name:
                        st.error("이름을 입력해주세요.")
                    elif any(u["name"] == new_name for u in users):
                        st.error("이미 등록된 이름입니다.")
                    else:
                        users.append({
                            "name": new_name,
                            "car_type": new_car,
                            "car_number": new_car_number if new_car_number else "",
                            "car_details": new_car_details if new_car_details else "",
                            "last_parked_date": None
                        })
                        save_json(USERS_FILE, users)
                        st.success(f"✅ {new_name}님이 추가되었습니다!")
                        st.rerun()
        
        st.divider()
        
        # Staff List with Excel Export
        if users:
            # Header with Excel button
            col_header, col_excel = st.columns([4, 1])
            with col_header:
                st.markdown("#### 등록된 직원")
            with col_excel:
                # Create DataFrame for export
                from io import BytesIO
                export_data = []
                for u in users:
                    export_data.append({
                        "이름": u["name"],
                        "차종": u["car_type"],
                        "차 번호": u.get("car_number", ""),
                        "상세 차종": u.get("car_details", ""),
                        "마지막 주차일": u.get("last_parked_date", "")
                    })
                
                df = pd.DataFrame(export_data)
                output = BytesIO()
                with pd.ExcelWriter(output, engine='openpyxl') as writer:
                    df.to_excel(writer, index=False, sheet_name='직원목록')
                excel_data = output.getvalue()
                
                st.download_button(
                    label="📥 엑셀",
                    data=excel_data,
                    file_name=f"직원목록_{datetime.now().strftime('%Y%m%d')}.xlsx",
                    mime="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
                    use_container_width=True
                )
            
            # Table-style list
            for idx, u in enumerate(users):
                col1, col2, col3, col4, col5, col6, col7 = st.columns([2, 1, 1.5, 1.5, 2, 0.4, 0.4])
                
                with col1:
                    st.write(f"**{u['name']}**")
                with col2:
                    st.write(u['car_type'])
                with col3:
                    st.write(u.get('car_number', '-'))
                with col4:
                    st.write(u.get('car_details', '-'))
                with col5:
                    st.write(u.get('last_parked_date', '기록 없음'))
                with col6:
                    if st.button("✏️", key=f"edit_user_{idx}"):
                        st.session_state[f"editing_user_{idx}"] = True
                        st.rerun()
                with col7:
                    if st.button("🗑️", key=f"del_user_{idx}"):
                        users.remove(u)
                        save_json(USERS_FILE, users)
                        st.rerun()
                
                # Edit form (if editing)
                if st.session_state.get(f"editing_user_{idx}", False):
                    with st.form(f"edit_form_{idx}"):
                        st.markdown("##### 정보 수정")
                        col1, col2 = st.columns(2)
                        edit_name = col1.text_input("이름", value=u["name"])
                        edit_car_type = col2.selectbox("차종", ["SEDAN", "SUV"], index=0 if u["car_type"] == "SEDAN" else 1)
                        
                        col3, col4 = st.columns(2)
                        edit_car_number = col3.text_input("차 번호", value=u.get("car_number", ""))
                        edit_car_details = col4.text_input("상세 차종", value=u.get("car_details", ""))
                        
                        col_save, col_cancel = st.columns(2)
                        with col_save:
                            if st.form_submit_button("💾 저장", type="primary"):
                                if edit_name != u["name"] and any(user["name"] == edit_name for user in users):
                                    st.error("이미 등록된 이름입니다.")
                                else:
                                    u["name"] = edit_name
                                    u["car_type"] = edit_car_type
                                    u["car_number"] = edit_car_number
                                    u["car_details"] = edit_car_details
                                    save_json(USERS_FILE, users)
                                    st.session_state[f"editing_user_{idx}"] = False
                                    st.success("✅ 저장되었습니다!")
                                    st.rerun()
                        with col_cancel:
                            if st.form_submit_button("❌ 취소"):
                                st.session_state[f"editing_user_{idx}"] = False
                                st.rerun()
                
                # Thin divider between rows
                if idx < len(users) - 1:
                    st.markdown("<hr style='margin: 2px 0; border: 0; border-top: 1px solid #e8e8ed;'>", unsafe_allow_html=True)
        else:
            st.info("등록된 직원이 없습니다.")
